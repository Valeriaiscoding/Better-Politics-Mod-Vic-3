labour_associations_events.4 = { # Stall
	type = country_event
	placement = root
	title = labour_associations_events.4.t
	desc = labour_associations_events.4.d
	flavor = labour_associations_events.4.f

	category = enactment

	duration = 3


	event_image = {
		video = "europenorthamerica_political_extremism"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	cooldown = { days = normal_modifier_time }

	trigger = {
		OR = {
            active_law:lawgroup_labour_associations ?= { has_amendment = amendment_type:bpm_amendment_artisan_guilds }
            active_law:lawgroup_labour_associations ?= { has_amendment = amendment_type:bpm_amendment_weakened_artisan_guilds }
        }
		OR = {
			is_enacting_law = law_type:law_combination_acts
			is_enacting_law = law_type:law_anti_strike_laws
			is_enacting_law = law_type:law_right_to_associate
			is_enacting_law = law_type:law_corporatized_unions
		}
		ig:ig_intelligentsia ?= {
			ig_counts_as_marginal = no
		}
		any_scope_state = {
			is_incorporated = yes
			turmoil >= 0.1
			any_scope_pop = {
				is_pop_type = shopkeepers
			}
			any_scope_building = {
				is_building_group = bg_manufacturing
			}
		}
	}

	immediate = {
		currently_enacting_law = {
    		save_scope_as = current_law_scope
    	}
		ig:ig_intelligentsia ?= {
			save_scope_as = pb_ig
		}
		random_scope_state = {
			limit = {
				is_incorporated = yes
				turmoil >= 0.1
				any_scope_pop = {
					is_pop_type = shopkeepers
				}
				any_scope_building = {
					is_building_group = bg_manufacturing
				}
			}
			save_scope_as = relevant_state
		}
	}

	cancellation_trigger = {
		NOT = { currently_enacting_law = scope:current_law_scope }
	}

	option = {
		name = labour_associations_events.4.a
		default_option = yes
		scope:relevant_state = {
			add_radicals_in_state = {
				pop_type = shopkeepers
				value = large_radicals
			}
			state_region = {
				add_devastation = 10
			}
		}
		add_enactment_modifier = {
			name = modifier_artisans_uprising
		}
	}

	option = {
		name = labour_associations_events.4.b
		scope:relevant_state = {
			add_radicals_in_state = {
				pop_type = shopkeepers
				value = large_radicals
			}
			state_region = {
				add_devastation = 5
			}
		}
		scope:pb_ig = {
			add_modifier = {
				name = modifier_repressed_artisans_riots
				days = normal_modifier_time
			}
		}
		add_enactment_modifier = {
			name = modifier_artisans_uprising
			multiplier = 0.35
		}
	}
}
