	add = bpm_additional_radicalism_factors_diplomacy

bpm_additional_radicalism_factors_regime = {
	if = {
		limit = {
			OR = {
				bpm_movement_is_socialist = yes
				bpm_movement_is_liberal_bourgeois = yes
			}
			owner = { 
				bpm_country_is_pre_constitutional = yes
			}
		}
		add = {
			desc = "IS_PRE_MODERN_ABSOLUTIST"
			value = 0.2
		}
	}
	else_if = {
		limit = {
			OR = {
				bpm_movement_is_socialist = yes
				bpm_movement_is_progressive_liberal = yes
			}
			owner = { 
				bpm_country_is_fascist = yes
			}
		}
		add = {
			desc = "ANTIFASCISM"
			value = 0.2
		}
	}
	else_if = {
		limit = {
			owner = {
				bpm_country_laws_are_very_undemocratic = yes
			}
			bpm_movement_is_liberal_bourgeois = yes
		}
		add = {
			desc = "VERY_UNDEMOCRATIC"
			value = 0.2
		}
	}
	else_if = {
		limit = {
			owner = { 
				bpm_country_laws_are_moderately_undemocratic = yes
			}
			bpm_movement_is_progressive_liberal = yes
		}
		add = {
			desc = "MODERATLY_UNDEMOCRATIC"
			value = 0.2
		}
	}

	if = {
		limit = {
			OR = {
				bpm_movement_is_modernizer = yes
				bpm_movement_is_socialist = yes
				bpm_movement_is_liberal_nationalist = yes
			}
			owner = { has_law_or_variant = law_type:law_extraction_economy }
		}
		add = {
			desc = "law_extraction_economy"
			value = 0.1
		}
	}
	if = {
		limit = {
			OR = {
				bpm_movement_is_modernizer = yes
			}
			owner = { country_has_laws_in_need_of_modernization = yes }
		}
		add = {
			desc = "WE_MUST_MODERNIZE"
			value = 0.2
		}
	}
	if = {
		limit = {
			OR = {
				bpm_movement_is_laissez_faire = yes
				bpm_movement_is_modernizer = yes
			}
			owner = { 
				bpm_country_laws_have_traditional_economics = yes
			}
		}
		add = {
			desc = "TRADITIONAL_ECONOMIC_SYSTEM"
			add = 0.1
		}
	}
	if = { # TODO Remove maybe
		limit = {
			bpm_movement_is_laissez_faire = yes
		}
		if = {
			limit = { owner = { bpm_country_laws_are_very_capitalist = no } }
			add = {
				desc = "NOT_FREE_MARKET" 
				value = 0.1
			}
		}
	}

	if = {
		limit = {
			bpm_movement_is_abolitionist = yes
			owner = { bpm_country_has_unfree_labor = yes }
		}
		add = {
			desc = "OPPOSES_SLAVERY"
			value = 0.2 
		}
	}
	if = {
		limit = {
			bpm_movement_is_slaver = yes
			owner = { 
				OR = {
					has_law_or_variant = law_type:law_legacy_slavery
					has_law_or_variant = law_type:law_slavery_banned
				}
			}
		}
		add = {
			desc = "SUPPORTS_SLAVERY"
			value = 0.2
		}
	}

	if = {
		limit = {
			OR = {
				bpm_movement_is_absolutist_monarchist = yes
				bpm_movement_is_liberal_bourgeois = yes
				is_political_movement_type = movement_fascist
				is_political_movement_type = movement_reactionary
			}
			owner = { 
				bpm_country_laws_has_socialist_economics = yes
			}
		}
		add = {
			desc = "OPPOSES_SOCIALISM"
			value = 0.2
		}
	}
	if = {
		limit = {
			OR = {
				bpm_movement_is_absolutist_monarchist = yes
				is_political_movement_type = movement_fascist
				is_political_movement_type = movement_reactionary
			}
			owner = { 
				OR = {
					bpm_country_laws_are_liberal_democratic = yes
					bpm_country_laws_are_moderately_liberal_democratic = yes
				} 
			}
		}
		add = {
			desc = "ANTI_DEMOCRATIC"
			value = 0.2
		}
	}



	# Governance Principle
	if = {
		limit = {
			bpm_movement_is_restorationist = yes
			owner = { bpm_country_is_republic = yes }
		}
		add = {
			desc = "MONARCHIST_MOVEMENT"
			value = 0.2
		}
	}
	else_if = {
		limit = {
			bpm_movement_is_absolutist_monarchist = yes
			owner = { bpm_country_is_constitutional_monarchy = yes }
		}
		add = {
			desc = "ABSOLUTE_MONARCHIST_MOVEMENT"
			value = 0.2
		}
	}
	else_if = {
		limit = {
			bpm_movement_is_anti_monarchy = yes
			owner = { bpm_country_is_republic = no }
			NOT = {
				is_political_movement_type = movement_anarchist
			}
			NAND = {
				is_political_movement_type = movement_communist
				owner = { has_variable = first_intl_chain_complete_var }
			}
		}
		add = {
			desc = "REPUBLICAN_MOVEMENT"
			value = 0.2
		}
	}

	# Misc
	if = {
		limit = {
			OR = {
				bpm_movement_is_liberal_nationalist = yes
				bpm_movement_is_national_supremacist = yes
			}
			owner.ruler ?= {
				NOT = { culture = { is_primary_culture_of = root.owner } }
			}
		}

		add = {
			value = 0.02
			desc = "RULER_NOT_PRIMARY_CULTURE"
		}
	}

}

bpm_additional_radicalism_factors_institutions = {
	if = {
		limit = {
			OR = {
				bpm_movement_is_liberal_bourgeois = yes
				bpm_movement_is_moderate_socialist = yes
				bpm_movement_is_modernizer = yes
				AND = {
					owner = { has_law_or_variant = law_type:law_council_republic }
					bpm_movement_is_radical_socialist = yes
				}
			}
		}
		owner = {
				if = {
					limit = { has_institution = institution_schools }
					
					add = {
						value = 3
						if = {
							limit = { root = { bpm_movement_is_radical_socialist = yes } }
							add = 1
						}
						subtract = "institution_investment_level(institution_schools)"
						multiply = 0.05
						desc = "INSTITUTION_FUNDING_LEVEL_schools"
					}
				}
				else = {
					add = {
						value = 0.1
						desc = "NO_INSTITUTION_schools"
					}		
				}
			}
	}
	if = {
		limit = {
			OR = {
				AND = {
					owner = { has_law_or_variant = law_type:law_council_republic }
					bpm_movement_is_radical_socialist = yes
				}
				bpm_movement_is_moderate_socialist = yes
			}
		}
		owner = {
			if = {
				limit = { has_institution = institution_social_security }
				
				add = {
					value = 3
					if = {
						limit = { root = { bpm_movement_is_radical_socialist = yes } }
						add = 1
					}
					subtract = "institution_investment_level(institution_social_security)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_social_security"
				}
			}
			else = {
				add = {
					value = 0.1
					if = {
						limit = { root = { bpm_movement_is_radical_socialist = yes } }
						value = 0.2
					}
					desc = "NO_INSTITUTION_social_security"
				}		
			}				
			
			if = {
				limit = { has_institution = institution_workplace_safety }
				
				add = {
					value = 3
					if = {
						limit = { root = { bpm_movement_is_radical_socialist = yes } }
						add = 1
					}
					subtract = "institution_investment_level(institution_workplace_safety)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_workplace_safety"
				}
			}
			else = {
				add = {
					value = 0.1
					if = {
						limit = { root = { bpm_movement_is_radical_socialist = yes } }
						value = 0.2
					}
					desc = "NO_INSTITUTION_workplace_safety"
				}		
			}
		}
	}
	if = {
		limit = {
			bpm_movement_is_modernizer = no
			bpm_movement_is_laissez_faire = yes
		}
		owner = {
			if = {
				limit = { 
					has_institution = institution_social_security 
					NOT = { has_law_or_variant = law_type:law_poor_laws }
				}
				
				add = {
					value = "institution_investment_level(institution_social_security)"
					multiply = 0.02
					desc = "INSTITUTION_FUNDING_LEVEL_LF_MOVEMENT_social_security"
				}
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_no_welfare
				}
				add = {
					value = 0.1
					desc = "NO_POOR_LAWS"
				}		
			}
			if = {
				limit = { 
					has_institution = institution_heathcare 
					NOT = { has_law_or_variant = law_type:law_private_health_insurance }
				}
				
				add = {
					value = "institution_investment_level(institution_health_system)"
					multiply = 0.02
					desc = "INSTITUTION_FUNDING_LEVEL_LF_MOVEMENT_health_system"
				}
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_no_health_system
				}
				add = {
					value = 0.1
					desc = "NO_PRIVATE_HEALTHCARE"
				}		
			}
			if = {
				limit = { 
					has_institution = institution_workplace_safety 
				}
				
				add = {
					value = "institution_investment_level(institution_workplace_safety)"
					multiply = 0.02
					desc = "INSTITUTION_FUNDING_LEVEL_workplace_safety"
				}
			}
			if = {
				limit = { 
					has_law = law_type:law_laissez_faire
				}
				
				add = {
					value = -1
					add = "institution_investment_level(institution_economy)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_economy"
				}
			}
			else = {
				add = {
					value = 0.1
					desc = "no_laissez_faire"
				}		
			}
		}
	}
	if = { # Maybe cut suffrage will be removed anyway
		limit = {
			owner = { 
				is_ai = no
				has_institution = institution_suffrage 
			}
		}
		owner = {
			if = {
				limit = {
					ROOT = {
						OR = {
							bpm_movement_is_moderate_socialist = yes
							AND = {
								bpm_movement_is_radical_socialist = yes
								owner = { has_law_or_variant = law_type:law_council_republic }
							}
							bpm_movement_is_progressive_liberal = yes
						}
					}
				}
				add = {
					value = 3
					subtract = "institution_investment_level(institution_suffrage)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_suffrage"
				}
			}
			else_if = {
				limit = {
					ROOT = { 
						OR = {
							bpm_movement_is_absolutist_monarchist = yes
							is_political_movement_type = movement_reactionary
							is_political_movement_type = movement_fascist
						}
					}
				}
				add = {
					add = "institution_investment_level(institution_suffrage)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_suffrage"
				}
			}
		}
		
	}

	if = {
		limit = {
			OR = {
				bpm_movement_is_liberal_nationalist = yes
				bpm_movement_is_modernizer = yes
			}
		}
		owner = {
			if = {
				limit = { 
					has_institution = institution_centralization
				}
				
				add = {
					value = 3
					subtract = "institution_investment_level(institution_centralization)"
					if = {
						limit = { is_ai = yes }
						min = 0
					}
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_centralization"
				}
			}
			else = {
				add = {
					value = 0.1
					desc = "NO_INSTITUTION_centralization"
				}		
			}
		}
	}

	if = {
		limit = {
			bpm_movement_is_liberal_nationalist = yes
			owner = {
				NOT = { has_law_or_variant = law_type:law_council_republic }
				bpm_country_laws_has_colonial_government = no
			}
		}
		owner = {		
			if = {
				limit = { 
					OR = {
						root = { bpm_movement_is_regressive = yes }
						bpm_country_laws_are_very_undemocratic = no
					}
				}
				if = {
					limit = { 
						has_institution = institution_police
					}
					
					add = {
						value = 3
						subtract = "institution_investment_level(institution_police)"
						multiply = 0.05
						desc = "INSTITUTION_FUNDING_LEVEL_police"
					}
				}
				else = {	
					add = {
						value = 0.1
						desc = "NO_INSTITUTION_police"
					}
				}
				if = {
					limit = { 
						has_institution = institution_culture
						OR = {
							NOT = { has_law_or_variant = law_type:law_outlawed_dissent }
							root = { bpm_movement_is_regressive = yes }
						}
					}
					
					add = {
						value = 3
						subtract = "institution_investment_level(institution_culture)"
						multiply = 0.05
						desc = "INSTITUTION_FUNDING_LEVEL_culture"
					}
				}
				else = {	
					add = {
						value = 0.1
						desc = "NO_INSTITUTION_culture"
					}
				}
				
			}
		}
	}	

	if = {
		limit = {
			bpm_movement_is_modernizer = yes
		}
		owner = {
			if = {
				limit = { 
					has_institution = institution_economy
					NOR = {
						has_law_or_variant = law_type:law_traditionalism
						has_law_or_variant = law_type:law_extraction_economy
					}
				}
				
				add = {
					value = 2
					subtract = "institution_investment_level(institution_economy)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_economy"
				}
			}
			
		}
	}
	if = {
		limit = {
			NOT = { is_political_movement_type = movement_reformist_socialist }
			bpm_movement_is_socialist = yes
			owner = { has_law = law_type:law_council_republic }
		}
		owner = {
			if = {
				limit = { 
					bpm_country_laws_has_socialist_economics = yes
					trigger_if = {
						limit = { ROOT = { is_political_movement_type = movement_anarchist } }
						has_law_or_variant = law_type:law_cooperative_ownership
					}
				}
				
				add = {
					value = -3
					add = "institution_investment_level(institution_economy)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_economy"
				}
			}
			else = {
				add = {
					value = 0.1
					desc = "NO_SOCIALIST_ECONOMY"
				}		
			}
			
		}
	}
}

bpm_additional_radicalism_factors_diplomacy = {
	# Catholic Movements and Pope
	if = {
		limit = {
			OR = {
				religion = rel:catholic
				is_political_movement_type = movement_miguelist
				is_political_movement_type = movement_carlist
				is_political_movement_type = movement_legitimist
				AND = {
					is_political_movement_type = movement_reactionary
					owner = { 
						OR = {
							country_has_state_religion = rel:catholic
							religion_percent_country = { 
								target = rel:catholic
								value >= 0.5
							}
						}
					}
				}
			}
			exists = c:PAP
			c:PAP ?= {
				country_has_state_religion = rel:catholic
				has_law_or_variant = law_type:law_theocracy
			}
			"owner.relations(c:PAP)" < 0
		}
		add = {
			desc = "POOR_RELATIONS_WITH_PAPACY"
			value = "owner.relations(c:PAP)"
			divide = 100
			multiply = -0.25
		}
	}
	# Subjects
	if = {
		limit = {
			owner ?= {
				is_subject = yes
			}
			OR = {
				bpm_movement_is_nationalist = yes
				AND = {
					is_political_movement_type = movement_cultural_minority
					owner = { has_law_or_variant = law_type:law_colonial_administration }
				}
				# AND = {
				# 	bpm_movement_is_radical_socialist = yes
				# 	owner = { NOT = { has_law = law_type:law_council_republic } }
				# }
			}
			NAND = { 
				owner = { has_law_or_variant = law_type:law_colonial_administration }
				is_political_movement_type = movement_cultural_majority
			}
			
		} 
		add = {
			desc = "[concept_subject]"
			if = {
				limit = {
					owner ?= {
						OR = {
							is_subject_type = subject_type_puppet
							is_subject_type = subject_type_crown_land
							is_subject_type = subject_type_vassal
						}
					}
				}
				value = 0.25
			}
			if = {
				limit = {
					owner ?= {
						OR = {
							is_subject_type = subject_type_tributary
							is_subject_type = subject_type_protectorate
							is_subject_type = subject_type_personal_union
						}
					}
				}
				value = 0.10
			}
		}
		add = {
			desc = "INDIA_SUBJECT"
			if = {
				limit = {
					owner ?= {
						OR = {
							is_subject_type = subject_type_colony
							is_subject_type = subject_type_chartered_company
						}
					}
				}
				value = 0.25
			}
			if = {
				limit = {
					owner ?= {
						is_subject_type = subject_type_dominion
					}
				}
				value = 0.10
			}
		}
		add = {
			desc = "LIBERTY_DESIRE"
			value = owner.liberty_desire
			divide = 500
		}
	}	

	if = {
		limit = {
			OR = {
				bpm_movement_is_modernizer = yes
				bpm_movement_is_nationalist = yes
			}
			owner = { is_country_type = unrecognized }
		}
		if = {
			limit = {
				owner = {
					has_variable = wargoal_enforced_by_recognized_power
				}
			}
			add = {
				desc = "wargoal_pressed_by_recognized_power_tt"
				value = 0.10
			}
		}
		if = {
			limit = {
				owner = {
					has_technology_researched = nationalism
				}
			}
			add = {
				desc = "unrecognized"
				value = 0.10
			}
		}
	}

}

				limit = {
					any_scope_state = {

								this ?= cu:north_german
								this ?= cu:south_german
							}
							has_homeland = prev
						}
						owner = { 
							NOR = {
								country_has_primary_culture = cu:south_german
								country_has_primary_culture = cu:north_german

						}
						
					}
				}
				add = 0.05
			}
			max = 0.25
		}
	}

		limit = { is_political_movement_type = movement_italian_national }

							OR = {
								this ?= cu:north_italian

							}
						}
						
					}
				}
				add = 0.05
			}
			max = 0.25
		}
	}

	if = {
		limit = { is_political_movement_type = movement_yugoslav_national }
		add = {
			desc = "Yugoslav [concept_homeland]s Under Foreign Control"
			every_state_region = {
				limit = {
					any_scope_state = {

							OR = {
								this ?= cu:serb
								this ?= cu:croat

							}

						}
						owner = { 
							NOR = {
								country_has_primary_culture = cu:serb
								country_has_primary_culture = cu:croat
								country_has_primary_culture = cu:bosniak
								country_has_primary_culture = cu:slovene
							}
						}
						
					}
				}
				add = 0.03
			}
			max = 0.3
		}
	}
}

bpm_additional_radicalism_factors_law_enactment = {
	if = {
		limit = {
			bpm_movement_is_restorationist = yes
			
		}
		owner ?= {
			if = {
				limit = {		
					has_law = law_type:law_monarchy
					trigger_if = {
						limit = { ROOT = { is_political_movement_type = movement_orleanist } }
						ruler = { has_variable = house_orleans }
					}
					trigger_else_if = {
						limit = { ROOT = { is_political_movement_type = movement_bonapartist } }
						ruler = { has_variable = house_bonaparte }
					}
				}
				if = {
					limit = {
						OR = {
							is_enacting_law = law_type:law_presidential_republic
							is_enacting_law = law_type:law_parliamentary_republic
							is_enacting_law = law_type:law_theocracy
							is_enacting_law = law_type:law_council_republic	
						}					
					}
					add = {
						if = {
							limit = {
								ROOT = { bpm_movement_is_absolutist_monarchist = yes }
							}
							value = 0.5
						}
						else = { value = 0.25 }
						desc = "MONARCHY_BEING_ABOLISHED"
					}
										
				}
			}		
		}
	}

	if = {
		limit = {
			OR = {
				bpm_movement_is_radical_socialist = yes
				is_political_movement_type = movement_socialist
			}
		}
		owner = {
			if = {
				limit = {
					has_law_or_variant = law_type:law_factory_councils
				}
				if = {
					limit = {
						OR = {
							country_is_abolishing_factory_councils = yes
							AND = {
								ROOT = { is_political_movement_type = movement_anarchist }
								is_enacting_law = law_type:law_socialist_labor_unions
							}
						}
					}
					add = {
						value = 0.25
						desc = "FACTORY_COUNCILS_BEING_ABOLISHED"
					}
				}
			}
		}
	}
}

bpm_additional_radicalism_factors_events = {
	if = {
		limit = {
			is_political_movement_type = movement_communist
			owner = { NOT = { has_law_or_variant = law_type:law_council_republic } }
		}
		if = {
			limit = {
				has_global_variable = second_intl_chain_complete_var
			}
			add = {
				desc = "COMMUNISTS_SECOND_INTERNATIONAL"
				value = 0.4 
			}
		}
		else_if = {
			limit = {
				has_global_variable = first_intl_chain_complete_var
			}
			add = {
				desc = "COMMUNISTS_FIST_INTERNATIONAL"
				value = 0.2
			}
		}
	}	
	if = {
		limit = { 
			is_political_movement_type = movement_cultural_minority
			owner = { has_journal_entry = je_springtime_of_the_peoples } 
			scope:culture = { has_discrimination_trait_group = heritage_group_european }
		}
		add = {
			value = 0.25
			desc = "SPRINGTIME_OF_THE_PEOPLES_UNDERWAY"
		}
	}	
}

	if = {
		limit = {
			NOR = {
				#is_political_movement_type = movement_cultural_majority
				#is_political_movement_type = movement_religious_majority
				#is_political_movement_type = movement_hindu_nationalist
				is_political_movement_type = movement_cultural_minority
				is_political_movement_type = movement_minority_rights
				is_political_movement_type = movement_religious_minority
				is_political_movement_type = movement_muslim_nationalist
			}
		}
		if = {
			limit = {
				political_movement_support <= 0.01
			}
			multiply = {
				value = 0.33
				desc = "VERY_LOW_MOVEMENT_SUPPORT"
			}
		}
		else_if = {
			limit = {
				political_movement_support <= 0.05
			}
			multiply = {

			}
		}
	}
