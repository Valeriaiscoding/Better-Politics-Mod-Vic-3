bpm_additional_radicalism_factor_ruler = {
	if = {
		limit = {
			OR = {
				is_political_movement_type = movement_slave_revolt
				is_political_movement_type = movement_anti_slavery
				is_political_movement_type = movement_utilitarian
			}
			owner.ruler ?= {
				OR = {
					has_ideology = ideology:ideology_jacksonian_democrat
					has_ideology = ideology:ideology_slaver
					AND = {
						NOR = {
							has_ideology = ideology:ideology_abolitionist
							has_ideology = ideology:ideology_utilitarian_leader
							has_ideology = ideology:ideology_utopian
						}
						interest_group = {
							bpm_ig_is_pro_slavery = yes
						}
					}
				}
			}
		}

		add = {
			value = 0.1
			desc = "RULER_IS_PRO_SLAVERY"
		}
	}
	else_if = {
		limit = {
			bpm_movement_is_slaver = yes
			owner.ruler ?= {
				OR = {
					has_ideology = ideology:ideology_abolitionist
					AND = {
						NOR = {
							has_ideology = ideology:ideology_slaver
							has_ideology = ideology:ideology_jacksonian_democrat
						}
						interest_group = { bpm_ig_is_abolitionist = ye }
					}
				}
			}
		}
		
		add = {
			value = 0.1
			desc = "RULER_IS_ANTI_SLAVERY"
		}
	}

	# Pissed rather than outright hateful, kept for flavor
	if = {
		limit = {
			OR = {
				bpm_movement_is_liberal_nationalist = yes
				bpm_movement_is_national_supremacist = yes
			}
			owner.ruler ?= {
				NOT = { culture = { is_primary_culture_of = root.owner } }
			}
		}

		add = {
			value = 0.02
			desc = "RULER_NOT_PRIMARY_CULTURE"
		}
	}
	if = {
		limit = {
			bpm_movement_is_moralist = yes
			trigger_if = {
				limit = { is_political_movement_type = movement_religious_majority }
				owner.ruler ?= {
					NOT = { religion = root.religion }
				}
			}
			trigger_else = {
				# Miguelists, Carlists, Legitimist, and Integralists
				owner.ruler ?= {
					NOT = { religion = rel:catholic }
				}
			}
		}		
		
		add = {
			value = 0.02
			desc = "RULER_WRONG_RELIGION"
		}			
	}
}

bpm_additional_radicalism_factors_law_enactment = {
	# Royalist hate abolition of monarchy
	# Bonapartists and Orleanists will care only if their monarchy is being abolished
	if = {
		limit = {
			bpm_movement_is_restorationist = yes
			
		}
		owner ?= {
			if = {
				limit = {		
					has_law = law_type:law_monarchy
					trigger_if = {
						limit = { ROOT = { is_political_movement_type = movement_orleanist } }
						ruler = { has_variable = house_orleans }
					}
					trigger_else_if = {
						limit = { ROOT = { is_political_movement_type = movement_bonapartist } }
						ruler = { has_variable = house_bonaparte }
					}
				}
				if = {
					limit = {
						OR = {
							is_enacting_law = law_type:law_presidential_republic
							is_enacting_law = law_type:law_parliamentary_republic
							is_enacting_law = law_type:law_theocracy
							is_enacting_law = law_type:law_council_republic	
						}					
					}
					add = {
						if = {
							limit = {
								ROOT = { bpm_movement_is_absolutist_monarchist = yes }
							}
							value = 0.5
						}
						else = { value = 0.25 }
						desc = "MONARCHY_BEING_ABOLISHED"
					}
										
				}
			}		
		}
	}
	# Abolition of factory councils
	if = {
		limit = {
			OR = {
				bpm_movement_is_radical_socialist = yes
				is_political_movement_type = movement_socialist
			}
		}
		owner = {
			if = {
				limit = {
					has_law_or_variant = law_type:law_factory_councils
				}
				if = {
					limit = {
						OR = {
							country_is_abolishing_factory_councils = yes
							AND = {
								ROOT = { is_political_movement_type = movement_anarchist }
								is_enacting_law = law_type:law_socialist_labor_unions
							}
						}
					}
					add = {
						value = 0.25
						desc = "FACTORY_COUNCILS_BEING_ABOLISHED"
					}
				}
			}
		}
	}
	
	if = {
		limit = { owner = { enacting_any_law = yes } }
		save_temporary_scope_as = polmov
		owner.currently_enacting_law = { save_temporary_scope_as = law }
		# Emblodened by enactment
		if = {
			limit = {
				law_enactment_stance = {
					law = scope:law.type
					value >= approve
				}
			}
			add = {
				desc = "BPM_EMBOLDENED_BY_LAW"
				value = movement_law_approval_to_number
				if = {
					limit = {  
						is_political_movement_type = movement_anarchist
						owner = { NOT = { has_law_or_variant = law_type:law_anarchy } }
					}
					multiply = 0.3
				}
				else_if = {
					limit = { 
						is_political_movement_type = movement_fascist
						owner = { bpm_country_is_fascist = no }
					}
					multiply = 0.3
				}
				else_if = {
					limit = { 
						is_political_movement_type = movement_communist
						owner = { NOT = { has_law_or_variant = law_type:law_council_republic } }
					}
					multiply = 0.2
				}
				else_if = {
					limit = { 
						bpm_movement_is_single_issue = yes
					}
					multiply = 0.2
				}
				else = {
					multiply = 0.1
				}

			}
		}
		# Radicalized by enactment
		else_if = {
			limit = {
				law_enactment_stance = {
					law = scope:law.type
					value <= disapprove
				}
			}
			add = { 
				desc = "Radicalized By Enactment"
				value = movement_law_approval_to_number
				if = {
					limit = {
						bpm_lawgroup_is_crucial_for_movement = yes
					}
					multiply = 0.2
				}
			}
		}
	}
}

				is_political_movement_type = movement_religious_majority
				is_political_movement_type = movement_cultural_minority
				is_political_movement_type = movement_minority_rights
				is_political_movement_type = movement_religious_minority
				is_political_movement_type = movement_muslim_nationalist
			}
		}
		if = {
			limit = {
				political_movement_support < 0.01
			}
			add = {
				value = -1
				desc = "NO_MOVEMENT_SUPPORT"
			}
		}
		else_if = {
			limit = {
				political_movement_support <= 0.01
			}
			add = {
				value = -0.25
				desc = "VERY_LOW_MOVEMENT_SUPPORT"
			}
		}
		else_if = {
			limit = {
				political_movement_support <= 0.05
			}
			add = {
				value = -0.1
				desc = "LOW_MOVEMENT_SUPPORT"
			}
		}
	}
