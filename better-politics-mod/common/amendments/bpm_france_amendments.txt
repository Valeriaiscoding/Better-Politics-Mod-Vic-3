bpm_amendment_artisanal_economy = {

	parent = law_industry_banned

	allowed_laws = {
		law_laissez_faire
		law_interventionism
		law_industry_banned
		law_agrarianism
	}

	modifier = {
		bpm_france_artisanal_lowest_level_pm = yes
		interest_group_ig_intelligentsia_pol_str_mult = 1
		country_shopkeepers_pol_str_mult = 0.75

		state_capitalists_investment_pool_efficiency_mult = -0.33

		country_production_tech_spread_mult = -0.25

		country_company_throughput_bonus_add = -0.15
		country_company_construction_efficiency_bonus_add = -0.4
		building_company_worker_dividends_add = 0.33

		building_group_bg_manufacturing_allowed_collectivization_add = 0.75
		building_group_bg_extraction_allowed_collectivization_add = 0.75
		building_group_bg_private_infrastructure_allowed_collectivization_add = 0.75
	}

	on_activate = {
		ig:ig_intelligentsia = {
			bpm_remove_all_eco_ideologies = yes
			add_ideology = ideology_eco_artisan
			bpm_reorder_ideologies = yes
		}
		ig:ig_petty_bourgeoisie = {
			bpm_remove_all_eco_ideologies = yes
			add_ideology = ideology_eco_artisan
			bpm_reorder_ideologies = yes
		}
		ig:ig_armed_forces = {
			bpm_remove_all_eco_ideologies = yes
			add_ideology = ideology_eco_artisan
			bpm_reorder_ideologies = yes
		}
	}

	on_deactivate = {
		ig:ig_intelligentsia = {
			if = {
				limit = { has_ideology = ideology:ideology_eco_artisan }
				bpm_remove_all_eco_ideologies = yes
				bpm_reorder_ideologies = yes
			}
		}
		ig:ig_petty_bourgeoisie = {
			if = {
				limit = { has_ideology = ideology:ideology_eco_artisan }
				bpm_remove_all_eco_ideologies = yes
				bpm_reorder_ideologies = yes
			}
		}
		ig:ig_armed_forces = {
			if = {
				limit = { has_ideology = ideology:ideology_eco_artisan }
				bpm_remove_all_eco_ideologies = yes
				bpm_reorder_ideologies = yes
			}
		}
	}

	possible = {
		c:FRA ?= this
	}

	would_sponsor = {
		is_interest_group_type = ig_intelligentsia
	}

	ai_will_revoke = {
		always = yes
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}
}
