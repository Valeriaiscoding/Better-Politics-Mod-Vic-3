bpm_country_carlist_movement_valid_trigger = {
    OR = {
        country_has_primary_culture = cu:spanish
        country_has_primary_culture = cu:catalan
        country_has_primary_culture = cu:basque
    }
    country_is_colonial_or_company = no
    NOT = {
        has_variable = monarchy_stamped_out_var
    }
    capital = {
        is_in_geographic_region = geographic_region_iberia
    }
}

bpm_country_legitimist_movement_valid_trigger = {
	OR = {
		c:FRA ?= this
		civil_war_origin_country ?= c:FRA
	}  
	NOT = { has_variable = monarchy_stamped_out_var }
	trigger_if = {
		limit = { has_variable = one_true_french_dynasty }
		NOR = {
			has_variable = bonapartist_restoration_var
			has_variable = orleanist_restoration_var
		}
	}
}

bpm_country_royalist_absolutist_valid_trigger = {
	bpm_country_carlist_movement_valid_trigger = no
	bpm_country_miguelist_movement_valid_trigger = no
	bpm_country_legitimist_movement_valid_trigger = no
	NOT = {
		has_variable = monarchy_stamped_out_var
	}
	OR = {
		country_has_voting_franchise = no
		AND = {
			NOT = {
				has_variable = king_in_parliament_possible
			}
			country_has_voting_franchise = yes
		}
	}
	NOT = {
		any_political_movement = {
			is_political_movement_type = movement_royalist_constitutional
		}
	}
}
