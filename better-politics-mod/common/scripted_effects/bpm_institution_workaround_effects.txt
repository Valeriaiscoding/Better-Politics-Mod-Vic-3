bpm_suffrage_institution_workaround = {
	if = {
		limit = {
			exists = root
			has_law = law_type:$VARIABLE$
		}
		add_modifier = { 
			name = institution_$VARIABLE$_workaround
			multiplier = {
				value = root.owner.institution:institution_suffrage.investment
			}
		}
	}
}

bpm_all_suffrage_institution_workaround = {
	bpm_remove_suffrage_workaround_modifiers = yes

		limit = {
			has_variable = bpm_minister_of_suffrage
			exists = var:bpm_minister_of_suffrage
		}
		if = {
			limit = {
				var:bpm_minister_of_suffrage.interest_group ?= {
					OR = {
						is_interest_group_type = ig_anarchists
						is_interest_group_type = ig_socialists
						is_interest_group_type = ig_revolutionist_socialists
						is_interest_group_type = ig_agrarian_populists

						is_interest_group_type = ig_radicals

				}

				name = institution_cabinet_suffrage_threshold_workaround

bpm_remove_centralization_workaround_modifiers = {
	remove_modifier = institution_law_confederated_workaround
	remove_modifier = institution_law_federated_workaround
	remove_modifier = institution_ai_centralization_workaround
	every_scope_state = {
		remove_modifier = institution_law_confederated_colony_workaround
		remove_modifier = institution_law_federated_colony_workaround
		remove_modifier = institution_law_devolved_colony_workaround
		remove_modifier = institution_law_unitary_colony_workaround
	}
}

bpm_centralization_institution_workaround = {
	if = {
		limit = {
			exists = root
			has_law = law_type:$VARIABLE$
		}
		add_modifier = { 
			name = institution_$VARIABLE$_workaround
			multiplier = {
				value = root.owner.institution:institution_centralization.investment
			}
		}
	}
}

bpm_centralization_institution_colony_workaround = {
	if = {
		limit = {
			exists = root
			has_law = law_type:$VARIABLE$
		}
		every_scope_state = {
			remove_modifier = institution_$VARIABLE$_colony_workaround
			if = {
				limit = { is_incorporated = no }
				add_modifier = { 
					name = institution_$VARIABLE$_colony_workaround
					multiplier = {
						value = 1
						subtract = incorporation_progress
						multiply = root.owner.institution:institution_centralization.investment
					}
				}
			}
		}
	}
}

bpm_centralization_institution_ai_balance = {
	if = {
		limit = { is_player = no }
		if = {
			limit = {
				exists = root
				institution_investment_level = { institution = institution_centralization value < 4 }
			}
			add_modifier = {
				name = institution_ai_centralization_workaround
				multiplier = {
					value = 4
					subtract = root.owner.institution:institution_centralization.investment
				}
			}
		}
	}
}

bpm_work_time_law_workaround = {
	if = {
		limit = { 
			is_building_group = $BG$
			bpm_building_can_use_slaves = no
		}	
		add_modifier = {
			name = bpm_worktime_$BG$_workaround
			multiplier = {
				value = 0.05
			}
		}
		add_modifier = {
			name = bpm_worktime_sol_workaround
			multiplier = modifier:building_bpm_modify_sol_per_workplace_safety_investment
		}
	}
}

bpm_all_work_time_law_workaround = {
	every_scope_state = {
		remove_modifier = bpm_worktime_institution_workaround_state
		remove_modifier = bpm_worktime_institution_slavery_workaround
		remove_modifier = bpm_worktime_institution_rural_workaround
		if = {
			limit = { is_slave_state = yes }
			add_modifier = {
				name = bpm_worktime_institution_slavery_workaround
				multiplier = {
					value = owner.modifier:country_bpm_incorporated_employment_mult
					multiply = incorporation_progress
				}
			}
		}
		else_if = {
			limit = {
				owner = { OR = { has_law_or_variant = law_type:law_serfdom has_law_or_variant = law_type:law_tenant_farmers } } 
			}
			add_modifier = {
				name = bpm_worktime_institution_rural_workaround
				multiplier = {
					value = owner.modifier:country_bpm_incorporated_employment_mult
					multiply = incorporation_progress
				}
			}
		}
		else = {
			add_modifier = {
				name = bpm_worktime_institution_workaround_state
				multiplier = {
					value = owner.modifier:country_bpm_incorporated_employment_mult
					multiply = incorporation_progress
				}
			}
		}
		every_scope_building ?= {
			bpm_remove_work_time_workaround_modifiers = yes
			bpm_work_time_law_workaround = { BG = bg_manufacturing }
			bpm_work_time_law_workaround = { BG = bg_service }
			bpm_work_time_law_workaround = { BG = bg_government }
			bpm_work_time_law_workaround = { BG = bg_infrastructure }
			bpm_work_time_law_workaround = { BG = bg_extraction }
			if = {
				limit = { 
					owner = { NOR = { has_law_or_variant = law_type:law_serfdom has_law_or_variant = law_type:law_tenant_farmers } }
				}
				bpm_work_time_law_workaround = { BG = bg_staple_crops }
				bpm_work_time_law_workaround = { BG = bg_ranching }
				bpm_work_time_law_workaround = { BG = bg_plantations }
			}
		}
		
	}
}
