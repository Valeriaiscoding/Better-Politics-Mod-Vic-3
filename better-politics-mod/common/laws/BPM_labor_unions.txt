# icon = graphical icon shown in-game

REPLACE:law_right_to_associate = {
	group = lawgroup_labour_associations
	
	icon = "gfx/interface/icons/law_icons/right_to_associate.dds"

	can_enact = {
		ig:ig_trade_unions = {
			is_marginal = no
		}
	}

	progressiveness = 50
	
	modifier = {
		interest_group_ig_trade_unions_pol_str_mult = 0.25
		country_bpm_auth_suppress_labor_disabled = yes
		country_populist_movement_support_add = 0.05
	}
	
	on_activate = {
		if = {
			limit = { game_date > 1836.1.2 }
			set_variable = union_law_introduced_var
		}
		ig:ig_trade_unions ?= {
			bpm_remove_all_dop_ideologies = yes
			add_ideology = ideology_dop_unionist
			bpm_reorder_ideologies = yes
		}
		if = {
			limit = { has_variable = bpm_state_union_activated }
			trigger_event = { id = bpm_union_leadership.2 }
		}
	}

	
	ai_enact_weight_modifier = {
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

REPLACE:law_factory_councils = {
	group = lawgroup_labour_associations

	icon = "gfx/interface/icons/law_icons/factory_councils.dds"

	progressiveness = 100

	unlocking_technologies = {
		socialism
	}

	# Intended that this does not require a Council Republic, this is putting the cart before the horse

	can_enact = {
		is_revolutionary = yes
		ig:ig_trade_unions = {
			is_marginal = no
		}
	}

	on_enact = {
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_right_to_associate
				}
			}
			custom_tooltip = trade_unions_much_more_radical_effect_tt
		}
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_guild_system
					has_law_or_variant = law_type:law_anti_strike_laws
					has_law_or_variant = law_type:law_corporatized_unions
				}
			}
			custom_tooltip = trade_unions_more_radical_effect_tt
		}
		if = {
			limit = {
					has_law_or_variant = law_type:law_corporatized_unions
			}
			custom_tooltip = trade_unions_less_corporatist_effect_tt
		}
	}

	modifier = {
		interest_group_ig_trade_unions_pol_str_mult = -0.75
		interest_group_ig_trade_unions_pop_attraction_mult = -0.75
		state_pop_support_movement_labor_mult = -0.10
		state_pop_support_movement_communist_mult = 0.20
		state_pop_support_movement_anarchist_mult = 0.20
		building_group_bg_manufacturing_allowed_collectivization_add = 0.5
		building_group_bg_extraction_allowed_collectivization_add = 0.5
		building_group_bg_power_allowed_collectivization_add = 0.5
		building_group_bg_private_infrastructure_allowed_collectivization_add = 0.5
		building_group_bg_arts_allowed_collectivization_add = 0.5
	}

	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
